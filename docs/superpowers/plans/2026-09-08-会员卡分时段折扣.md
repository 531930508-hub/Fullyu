> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 后台可配置「门店×卡类型」的会员卡分时段折扣规则（星期/时间段/日期范围组合，支持百分比折扣与按时长固定价），下单询价按预约开始时间自动匹配，未命中回退现有课程固定折扣。

/**
 * 会员卡分时段折扣共享定义。
 *
 * 门店按忙闲时段差异化定价：峰值时段取消折扣甚至加价、闲时深折引流。
 * 命中判定放在这里，保证后端下单锁价与后台"一周总览预览"算出同一个答案
 * （同审批流引擎 pickFlowRule 的模式）。
 */

/** 折扣动作类型（与 Prisma TimeDiscountActionType 对齐） */
export type TimeDiscountActionType = 'PERCENT' | 'FIXED_BY_DURATION'

export const TIME_DISCOUNT_ACTION_META: Record<TimeDiscountActionType, { text: string; desc: string }> = {
  PERCENT: { text: '百分比折扣', desc: '50–200；100=原价，<100 打折，>100 峰时加价' },
  FIXED_BY_DURATION: { text: '按时长固定价', desc: '按服务时长分钟数取固定价（如 30 分钟 = 133 元）' },
}

/** 一天内时间段，"HH:mm"，含头不含尾，start < end（不支持跨午夜） */
export interface TimeDiscountRange {
  start: string
  end: string
}

/** 时长固定价条目；price 单位：元（与全系统运行时 Number 元口径一致） */
export interface TimeDiscountFixedPrice {
  minutes: number
  price: number
}

export interface TimeDiscountRuleLike {
  id: string
  name?: string
  /** 适用门店 id；空数组 = 全部门店（兜底，新店自动继承） */
  orgIds: string[]
  /** 适用卡类型；空数组 = 全部卡类型 */
  cardTypes: string[]
  /** 越小越优先（同特异性分组内） */
  priority: number
  /** 1=周一 … 7=周日；空数组 = 不限 */
  weekdays: number[]
  /** 空数组 = 全天 */
  timeRanges: TimeDiscountRange[]
  /** 生效日期范围（含两端，按自然日）；null = 不限 */
  validFrom: string | Date | null
  validTo: string | Date | null
  actionType: TimeDiscountActionType
  /** PERCENT 时必填，50–200 */
  discountPercent: number | null
  /** FIXED_BY_DURATION 时必填 */
  fixedPrices: TimeDiscountFixedPrice[] | null
  enabled: boolean
}

export interface TimeDiscountHit {
  ruleId: string
  ruleName: string
  actionType: TimeDiscountActionType
  discountPercent: number | null
  /** 单位：元 */
  fixedPrice: number | null
}

export interface TimeDiscountContext {
  /** 订单归属门店；null（场地未挂门店）时只能命中全局规则 */
  orgId: string | null
  /** 用户当前有效会员卡类型 */
  cardType: string
  /** 本单预约开始时间（判定口径） */
  appointmentStart: Date
  /** 本单服务时长（分钟）；null 时 FIXED_BY_DURATION 规则无法命中 */
  durationMinutes: number | null
}

/** "HH:mm" → 当日分钟数；非法格式返回 null */
export function parseHHmm(v: string): number | null {
  const m = /^(\d{1,2}):(\d{2})$/.exec(String(v ?? '').trim())
  if (!m) return null
  const h = Number(m[1])
  const mi = Number(m[2])
  if (h > 23 || mi > 59) return null
  return h * 60 + mi
}

/** Date → 1=周一 … 7=周日（JS getDay 0=周日 转换） */
export function weekdayOf(d: Date): number {
  return ((d.getDay() + 6) % 7) + 1
}

/**
 * 日期字面量比较值 YYYYMMDD。
 * 字符串（含 Prisma 返回的 ISO 串）直接取前 10 位日期字面量，避免 UTC/本地时区偏移；
 * Date 对象按本地时区取年月日（部署时区 Asia/Shanghai，UTC 午夜 = 当地同日 08:00，无跨日问题）。
 */
function toDateKey(v: string | Date): number {
  if (typeof v === 'string') {
    const m = /^(\d{4})-(\d{2})-(\d{2})/.exec(v)
    if (m) return Number(m[1]) * 10000 + Number(m[2]) * 100 + Number(m[3])
  }
  const d = v instanceof Date ? v : new Date(v)
  return d.getFullYear() * 10000 + (d.getMonth() + 1) * 100 + d.getDate()
}

/** 规则排序：显式门店在前（特异性高），组内 priority 升序 */
export function sortTimeDiscountRules(rules: TimeDiscountRuleLike[]): TimeDiscountRuleLike[] {
  return [...rules].sort((a, b) => {
    const sa = a.orgIds.length > 0 ? 0 : 1
    const sb = b.orgIds.length > 0 ? 0 : 1
    if (sa !== sb) return sa - sb
    return a.priority - b.priority
  })
}

/**
 * 匹配时段折扣规则；全部未命中返回 null（调用方回退课程固定折扣 → 原价）。
 * FIXED_BY_DURATION 规则在服务时长未知或价目表不含该时长时视为未命中，继续匹配下一条。
 */
export function pickTimeDiscountRule(
  rules: TimeDiscountRuleLike[],
  ctx: TimeDiscountContext,
): TimeDiscountHit | null {
  const start = ctx.appointmentStart
  const dayKey = toDateKey(start)
  const wd = weekdayOf(start)
  const minuteOfDay = start.getHours() * 60 + start.getMinutes()

  for (const r of sortTimeDiscountRules(rules)) {
    if (!r.enabled) continue
    if (r.orgIds.length > 0 && (ctx.orgId == null || !r.orgIds.includes(ctx.orgId))) continue
    if (r.cardTypes.length > 0 && !r.cardTypes.includes(ctx.cardType)) continue
    if (r.validFrom != null && dayKey < toDateKey(r.validFrom)) continue
    if (r.validTo != null && dayKey > toDateKey(r.validTo)) continue
    if (r.weekdays.length > 0 && !r.weekdays.includes(wd)) continue
    if (r.timeRanges.length > 0) {
      const inRange = r.timeRanges.some((tr) => {
        const s = parseHHmm(tr.start)
        const e = parseHHmm(tr.end)
        return s != null && e != null && s < e && minuteOfDay >= s && minuteOfDay < e
      })
      if (!inRange) continue
    }
    if (r.actionType === 'PERCENT') {
      if (r.discountPercent == null) continue
      return {
        ruleId: r.id,
        ruleName: r.name ?? '',
        actionType: 'PERCENT',
        discountPercent: r.discountPercent,
        fixedPrice: null,
      }
    }
    if (ctx.durationMinutes == null || !r.fixedPrices) continue
    const fp = r.fixedPrices.find((p) => p.minutes === ctx.durationMinutes)
    if (!fp) continue
    return {
      ruleId: r.id,
      ruleName: r.name ?? '',
      actionType: 'FIXED_BY_DURATION',
      discountPercent: null,
      fixedPrice: fp.price,
    }
  }
  return null
}

      orderBy: [{ priority: 'asc' }, { createdAt: 'asc' }],
    })
  }

  async create(input: TimeDiscountInput) {
    this.validate(input)
    const max = await this.prisma.memberTimeDiscountRule.aggregate({ _max: { priority: true } })
    return this.prisma.memberTimeDiscountRule.create({
      data: {
        ...this.toData(input),
        priority: (max._max.priority ?? -1) + 1,
      },
    })
  }

  async update(id: string, input: TimeDiscountInput) {
    this.validate(input)
    const exist = await this.prisma.memberTimeDiscountRule.findUnique({ where: { id } })
    if (!exist) throw new BusinessException('时段折扣规则不存在', 4040)
    return this.prisma.memberTimeDiscountRule.update({ where: { id }, data: this.toData(input) })
  }

  async setStatus(id: string, status: 'ENABLED' | 'DISABLED') {
    const exist = await this.prisma.memberTimeDiscountRule.findUnique({ where: { id } })
    if (!exist) throw new BusinessException('时段折扣规则不存在', 4040)
    return this.prisma.memberTimeDiscountRule.update({ where: { id }, data: { status } })
  }

  /** 调序：传入完整 id 顺序，重排 priority = 下标 */
  async reorder(ids: string[]) {
    if (!Array.isArray(ids) || ids.length === 0) throw new BusinessException('参数错误', 4000)
    const total = await this.prisma.memberTimeDiscountRule.count()
    if (ids.length !== total) throw new BusinessException('排序需包含全部规则，请刷新后重试', 4000)
    await this.prisma.$transaction(
      ids.map((id, i) =>
        this.prisma.memberTimeDiscountRule.update({ where: { id }, data: { priority: i } }),
      ),
    )
    return { success: true }
  }

  async remove(id: string) {
    const exist = await this.prisma.memberTimeDiscountRule.findUnique({ where: { id } })
    if (!exist) throw new BusinessException('时段折扣规则不存在', 4040)
    const used = await this.prisma.order.count({ where: { hitTimeRuleId: id } })
    if (used > 0) throw new BusinessException('该规则已被订单引用，不可删除，请改为停用', 4000)
    await this.prisma.memberTimeDiscountRule.delete({ where: { id } })
    return { success: true }
  }

  private toData(input: TimeDiscountInput) {
    return {
      name: input.name.trim(),
      orgIds: input.orgIds ?? [],
      cardTypes: input.cardTypes ?? [],
      weekdays: input.weekdays ?? [],
      timeRanges: input.timeRanges ?? [],
      validFrom: input.validFrom ? new Date(input.validFrom) : null,
      validTo: input.validTo ? new Date(input.validTo) : null,
      actionType: input.actionType,
      discountPercent: input.actionType === 'PERCENT' ? input.discountPercent ?? null : null,
      fixedPrices: input.actionType === 'FIXED_BY_DURATION' ? input.fixedPrices ?? null : null,
      remark: input.remark ?? null,
    }
  }

  private validate(input: TimeDiscountInput) {
    if (!input.name?.trim()) throw new BusinessException('规则名称不能为空', 4000)
    const weekdays = input.weekdays ?? []
    if (weekdays.some((d) => !Number.isInteger(d) || d < 1 || d > 7)) {
      throw new BusinessException('星期取值须为 1-7', 4000)
    }
    for (const r of input.timeRanges ?? []) {
      const s = parseHHmm(r.start)
      const e = parseHHmm(r.end)
      if (s == null || e == null) throw new BusinessException('时间段格式须为 HH:mm', 4000)
      if (s >= e) throw new BusinessException('时间段开始须早于结束（不支持跨午夜）', 4000)
    }
    if (input.validFrom && input.validTo && new Date(input.validFrom) > new Date(input.validTo)) {
      throw new BusinessException('日期范围开始不能晚于结束', 4000)
    }
    if (input.actionType === 'PERCENT') {
      const p = input.discountPercent
      if (p == null || !Number.isFinite(p) || p < 50 || p > 200) {
        throw new BusinessException('折扣率须在 50–200 之间', 4000)
      }
    } else if (input.actionType === 'FIXED_BY_DURATION') {
      const fps = input.fixedPrices ?? []
      if (fps.length === 0) throw new BusinessException('按时长固定价至少需要一条价目', 4000)
      const seen = new Set<number>()
      for (const fp of fps) {
        if (!Number.isInteger(fp.minutes) || fp.minutes <= 0) {
          throw new BusinessException('时长（分钟）须为正整数', 4000)
        }
        if (seen.has(fp.minutes)) throw new BusinessException(`时长 ${fp.minutes} 分钟重复`, 4000)
        seen.add(fp.minutes)
        if (fp.price == null || !Number.isFinite(fp.price) || fp.price < 0) {
          throw new BusinessException('固定价须为非负数', 4000)
        }
      }
    } else {
      throw new BusinessException('未知的折扣动作类型', 4000)
    }
  }
}

import { Injectable } from '@nestjs/common'
import { PrismaService } from '../../common/prisma.service'
import { Prisma, MemberCardType } from '@prisma/client'
import {
  pickTimeDiscountRule,
  type TimeDiscountFixedPrice,
  type TimeDiscountHit,
  type TimeDiscountRange,
  type TimeDiscountRuleLike,
} from '@fullyu/shared'

export interface PriceQuote {
  position: number
  basePrice: number
  memberCardType: MemberCardType | null
  discountPercent: number | null
  discountAmount: number
  priceAfterDiscount: number
}

export interface MemberDiscountResult {
  cardType: MemberCardType | null
  discountPercent: number | null
  discountAmount: number
  /** 命中的时段折扣规则 id（未命中为 null） */
  hitTimeRuleId: string | null
  /** 命中详情（前端展示"会员时段价"用；未命中为 null） */
  timeDiscount: TimeDiscountHit | null
}

/**
 * 定价服务：阶梯价按报名顺序区间锁价 + 会员折扣计算。
 * 会员折扣 = 时段折扣规则（门店特异性 + 手动优先级）→ 课程固定折扣 → 无折扣。
 * 被报名下单与候补入队复用。
 */
@Injectable()
export class PricingService {
  constructor(private readonly prisma: PrismaService) {}

  /**
   * 计算课程下一个报名顺序位（已报名 + 待支付占位 + 1）
   */
  async getNextPosition(
    client: Prisma.TransactionClient | PrismaService,
    courseId: string,
    currentEnrolled: number,
  ): Promise<number> {
    const pending = await client.order.count({
      where: { courseId, orderType: 'COURSE', status: 'PENDING_PAYMENT' },
    })
    return currentEnrolled + pending + 1
  }

  /**
   * 按报名顺序位取阶梯价；无匹配区间时回退课程原价
   */
  getTierPrice(
    tiers: Array<{ seatFrom: number; seatTo: number; price: Prisma.Decimal | number }>,
    position: number,
    fallback: Prisma.Decimal | number,
  ): number {
    const hit = tiers.find((t) => position >= t.seatFrom && position <= t.seatTo)
    return Number(hit ? hit.price : fallback)
  }

  /** 由 startDate（日期）+ startTime（HH:mm）拼出课程实际开课时刻（同 checkin 模块口径） */
  courseStartAt(course: { startDate: Date; startTime: string }): Date {
    const d = new Date(course.startDate)
    const [h, m] = (course.startTime || '00:00').split(':').map((v) => Number(v))
    d.setHours(Number.isFinite(h) ? h : 0, Number.isFinite(m) ? m : 0, 0, 0)
    return d
  }

  /**
   * 计算会员折扣：
   * 1) 时段折扣规则（按预约开始时间判定，门店特异性 > 全局兜底，组内 priority 升序）
   * 2) 未命中 → 课程 × 卡类型固定折扣（原有逻辑）
   * 3) 都没有 → 无折扣
   * @returns discountAmount 按 basePrice 计算；>100% 加价或固定价高于原价时为负数
   */
  async getMemberDiscount(
    userId: string,
    courseId: string,
    basePrice: number,
    appointmentStart: Date,
  ): Promise<MemberDiscountResult> {
    const now = new Date()
    const card = await this.prisma.memberCard.findFirst({
      where: { userId, status: 'ACTIVE', endDate: { gt: now } },
      orderBy: { endDate: 'desc' },
    })
    if (!card) {
      return { cardType: null, discountPercent: null, discountAmount: 0, hitTimeRuleId: null, timeDiscount: null }
    }

    // 1) 时段折扣规则
    const course = await this.prisma.course.findUnique({
      where: { id: courseId },
      select: { duration: true, venue: { select: { orgId: true } } },
    })
    const rows = await this.prisma.memberTimeDiscountRule.findMany({ where: { status: 'ENABLED' } })
    const rules: TimeDiscountRuleLike[] = rows.map((r) => ({
      id: r.id,
      name: r.name,
      orgIds: (r.orgIds as string[]) ?? [],
      cardTypes: (r.cardTypes as string[]) ?? [],
      priority: r.priority,
      weekdays: (r.weekdays as number[]) ?? [],
      timeRanges: (r.timeRanges as TimeDiscountRange[]) ?? [],
      validFrom: r.validFrom,
      validTo: r.validTo,
      actionType: r.actionType,
      discountPercent: r.discountPercent,
      fixedPrices: (r.fixedPrices as TimeDiscountFixedPrice[] | null) ?? null,
      enabled: true,
    }))
    const hit = pickTimeDiscountRule(rules, {
      orgId: course?.venue?.orgId ?? null,
      cardType: card.type,
      appointmentStart,
      durationMinutes: course?.duration ?? null,
    })
    if (hit) {
      if (hit.actionType === 'PERCENT') {
        // discountPercent=70 表示 7 折；>100 时 discountAmount 为负 = 峰时加价
        const discountAmount = basePrice * (1 - (hit.discountPercent as number) / 100)
        return {
          cardType: card.type,
          discountPercent: hit.discountPercent,
          discountAmount,
          hitTimeRuleId: hit.ruleId,
          timeDiscount: hit,
        }
      }
      // FIXED_BY_DURATION：会员价 = 固定价，折扣额 = 原价与固定价之差（可为负）
      const fixedPrice = hit.fixedPrice as number
      return {
        cardType: card.type,
        discountPercent: null,
        discountAmount: basePrice - fixedPrice,
        hitTimeRuleId: hit.ruleId,
        timeDiscount: hit,
      }
    }

    // 2) 回退：课程 × 卡类型固定折扣（原有逻辑）
    const discount = await this.prisma.courseMemberDiscount.findUnique({
      where: { courseId_cardType: { courseId, cardType: card.type } },
    })
    if (!discount) {
      return { cardType: card.type, discountPercent: null, discountAmount: 0, hitTimeRuleId: null, timeDiscount: null }
    }
    const discountAmount = Math.max(0, basePrice * (1 - discount.discountPercent / 100))
    return {
      cardType: card.type,
      discountPercent: discount.discountPercent,
      discountAmount,
      hitTimeRuleId: null,
      timeDiscount: null,
    }
  }

  /**
   * 用户当前积分余额（取最近一条积分流水的余额）
   */
  async getPointsBalance(userId: string): Promise<number> {
    const last = await this.prisma.pointsRecord.findFirst({
      where: { userId },
      orderBy: { createdAt: 'desc' },
    })
    return last?.balance ?? 0
  }
}

  for (const tr of f.timeRanges) {
    if (!tr.start || !tr.end) return '时间段需填写完整'
    if (tr.start >= tr.end) return `时间段 ${tr.start}–${tr.end} 开始须早于结束`

  if (f.validFrom && f.validTo && f.validFrom > f.validTo) return '日期范围开始不能晚于结束'

    if (f.discountPercent == null || f.discountPercent < 50 || f.discountPercent > 200) return '折扣率须在 50–200 之间'

      if (seen.has(fp.minutes)) return `时长 ${fp.minutes} 分钟重复`
      seen.add(fp.minutes)
      if (fp.price == null || fp.price < 0) return '固定价须为非负数'
